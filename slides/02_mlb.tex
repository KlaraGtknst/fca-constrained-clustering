%%%% 2: MLB in FCA
\begin{frame}{MLB Constraints for Concept Lattices}
\hypertarget{main:mlb-constraints}{}
\begin{itemize}
    \item \acs{mlb} constraints $\Sigma \subseteq G \times G \times G$
    \begin{itemize}
        \item $MLB_{a b c}=(a,b,c) \in \Sigma$ {\tiny[\hyperlink{appendix:mlb-defs}{Formal properties}]}
    \end{itemize}

    \item<2-> Structural constraint saturation
    \begin{itemize}
       
        
        \item Exhaustively applying transitive inference rules {\tiny[\hyperlink{appendix:saturation}{Structural saturation}]}
        \begin{center}
            \begin{tikzpicture}[
                node distance=1.25cm,
                box/.style={draw,rounded corners=2pt,align=center,minimum width=2.7cm,minimum height=0.75cm},
                arr/.style={-{Latex[length=2mm]},thick}
            ]
            \node[box] (i1) {$(a, b, c)$};
            \node[box,below=0.05cm of i1] (i2) {$(b, c, d)$};
            \node[box,right=1.8cm of $(i1)!0.5!(i2)$] (i3) {$(a, b, d)$};
            \coordinate (i3upper) at ($(i3.west)!0.25!(i3.north west)$);
            \coordinate (i3lower) at ($(i3.west)!0.25!(i3.south west)$);
            \draw[arr] (i1.east) -- (i3upper);
            \draw[arr] (i2.east) -- (i3lower);
            \end{tikzpicture}
        \end{center}
    \end{itemize}

    \item<3-> Translate hierarchical \ac{mlb} constraints into implications:
        \[
        (a,b,c)
        \quad\Rightarrow\quad
        \{a,c\}\subseteq A \Rightarrow b\in A
        \quad\Leftrightarrow\quad
        a\wedge c\rightarrow b,
        \]
        and analogously $b\wedge c\rightarrow a$.

    \item<4-> Closure operator $\phi_{\Sigma}: \mathcal{P}(G) \to \mathcal{P}(G)$ {\tiny[\hyperlink{appendix:closure-proof}{Proof}]}
    \begin{itemize}
        \item $A \subseteq G$ is $\Sigma$-consistent iff
            $
            \forall (a,b,c)\in\Sigma:
            \{a,c\}\subseteq A \Rightarrow b\in A
            $
        \item $\phi_{\Sigma}(A)$ is the smallest $\Sigma$-consistent set containing $A$
    \end{itemize}

\end{itemize}
