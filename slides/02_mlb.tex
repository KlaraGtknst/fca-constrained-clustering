%%%% 2: MLB in FCA
\begin{frame}{MLB Constraints for Concept Lattices}
\hypertarget{main:mlb-constraints}{}
\begin{itemize}
    \item \acs{mlb} constraints $\Sigma \subseteq G \times G \times G$
    \begin{itemize}
        \item $MLB_{a b c}=(a,b,c) \in \Sigma$ {\tiny[\hyperlink{appendix:mlb-defs}{Formal properties}]}
    \end{itemize}

    \item<2-> $\forall (a,b,c) \in \Sigma: \{a, c\} \in A \rightarrow b \in A \Rightarrow A \subseteq G$ is $\Sigma$-consistent 

    \item<3-> Closure operator $\phi_{\Sigma}(A): \mathcal{P}(G) \to \mathcal{P}(G)$ maps $A$ to the smallest $\Sigma$-consistent set containing $A$ {\tiny[\hyperlink{appendix:closure-proof}{Proof}]}
    
    \item<4-> Structural constraint saturation
    \begin{itemize}
        \item Translate hierarchical \ac{mlb} constraints into implications before applying $\phi_{\Sigma}$:\\
        $(a,b,c)$ becomes $a \wedge c \Rightarrow b$ and $b \wedge c \Rightarrow a$
        \item Exhaustively applying transitive inference rules {\tiny[\hyperlink{appendix:saturation}{Structural saturation}]}
        \begin{center}

    \end{itemize}    
\end{itemize}
