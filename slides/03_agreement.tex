\begin{frame}{Consistency Lattice $\mathcal{C}_{\cap}$}
\hypertarget{main:consistency-overview}{}

    Let $\mathcal{C}_{\Sigma}$ be the \ac{mlb} closure system and $\mathcal{C}_{TM}$ the topic-model closure system. 

\[
\mathcal{C}_{\cap}
= \mathcal{C}_{\Sigma}\cap\mathcal{C}_{TM}
= \{A\subseteq G \mid \phi_{\Sigma}(A)=A \land \phi_{TM}(A)=A\}
\]

\begin{columns}[T,onlytextwidth]
\begin{column}{0.48\textwidth}
\begin{itemize}
    \item<2-> Consistent concept: Extent is closed in both closure systems
    \item[$\rightarrow$]<3-> Requires identical documents in extents (i.e., exact agreement)
    \vspace{0.5em}
    \item[\danger]<4-> BankSearch: Only trivial shared concepts remain. Exact agreement is too strict for this practical setting.
\end{itemize}

\end{column}

\begin{frame}{Coherence Lattice $\mathcal{C}_{coh}$}
  
Let $\mathcal{C}_{\Sigma}$ be the \ac{mlb} closure system and $\mathcal{C}_{TM}$ the topic-model closure system. 

\[
C_{coh} = \{ A_1 \cap A_2 \mid A_1 \in \mathcal{C}_{\Sigma}, A_2 \in \mathcal{C}_{TM}, J(A_1, A_2) > \theta \}
\]
\begin{columns}[T,onlytextwidth]
\begin{column}{0.55\textwidth}
\begin{itemize}
    \item<2-> Concepts $(A_1, B_1) \in \mathcal{C}_{\Sigma}, (A_2, B_2) \in \mathcal{C}_{TM}$
    \item<3-> Jaccard similarity~\cite{Glckstad2013UnsupervisedKS} {\tiny[\hyperlink{appendix:agreement-heatmap}{Agreement heatmap}]}:
        \[
        J(A_1,A_2)=\frac{|A_1\cap A_2|}{|A_1\cup A_2|}
        \]
        
\end{itemize}
\end{column}
