\begin{frame}{Step 3: Build the Explainable Data-Driven Hierarchy TODO}
\begin{itemize}
    \item Train \ac{lda} with $T=11$ topics to match the represented BankSearch categories.
    \item Convert document-topic probabilities into a binary context with threshold $\delta_t=0.07$.
    \item Apply an iceberg support threshold $\delta_{ms}=0.05$ to retain interpretable frequent concepts.
    \item The resulting \ac{fca} concepts are explained by their topic intents.
\end{itemize}

\begin{frame}{Consistency Lattice $\mathcal{C}_{\cap}$}
    Let $\mathcal{C}_{\Sigma}$ be the \ac{mlb} closure system and $\mathcal{C}_{TM}$ the topic-model closure system. 

\[
\mathcal{C}_{\cap}
= \mathcal{C}_{\Sigma}\cap\mathcal{C}_{TM}
= \{A\subseteq G \mid \phi_{\Sigma}(A)=A \land \phi_{TM}(A)=A\}
\]

\begin{columns}[c,onlytextwidth]
\begin{column}{0.48\textwidth}
\begin{itemize}
    \item Consistent concept: Extent is closed in both closure systems
    \item[$\rightarrow$] Requires identical documents in extents (i.e., exact agreement)
    \vspace{0.5em}
    \item[\danger] BankSearch: Only trivial shared concepts remain. Exact agreement is too strict for this practical setting.
\end{itemize}
\end{column}

\begin{frame}{Coherence Lattice $\mathcal{C}_{coh}$}
  
Let $\mathcal{C}_{\Sigma}$ be the \ac{mlb} closure system and $\mathcal{C}_{TM}$ the topic-model closure system. 
\begin{columns}[c,onlytextwidth]
\begin{column}{0.55\textwidth}
\begin{itemize}
    \item Concepts $(A_1, B_1) \in \mathcal{C}_{\Sigma}, (A_2, B_2) \in \mathcal{C}_{TM}$,
    \item Comparison using Jaccard similarity~\cite{Glckstad2013UnsupervisedKS}, {\tiny[\hyperlink{appendix:agreement-heatmap}{Agreement heatmap}]}:
        \[
        J(A_1,A_2)=\frac{|A_1\cap A_2|}{|A_1\cup A_2|}
        \]
    \item Coherence lattice $\mathcal{C}_{coh}$:
        \[
        M_{coh} = \{ A_1 \cap A_2 \mid A_1 \in \mathcal{C}_{\Sigma}, A_2 \in \mathcal{C}_{TM}, J(A_1, A_2) > \theta \}
        \]
\end{itemize}
\end{column}
