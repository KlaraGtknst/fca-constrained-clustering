%%%% 5: Manual vs. Data-Driven Hierarchy
\begin{frame}{Source of Background Knowledge}

\begin{columns}[T,onlytextwidth]
\begin{column}{0.48\textwidth}
\textbf{Manual Annotation}
\begin{itemize}
    \item[+]<2-> Trust-worthy expert knowledge
    \item[--]<3-> Snapshot of evolving domain knowledge
    \item[--]<4-> costly, time-consuming
\end{itemize}
\end{column}

\begin{column}{0.48\textwidth}
\textbf{Data-Driven}
\begin{itemize}
    \item[--]<2-> Quality depends on data
    \item[+]<3-> Extends coverage of domains
    \item[+]<4-> Less costly, less time-consuming
\end{itemize}
\end{column}
\end{columns}
