%%%% 4: Manual vs. Data-Driven Hierarchy
\begin{frame}{Source of Background Knowledge}

\only<-4>{
Existing data-driven approaches use structured knowledge~\cite{Priss:2004,old:04,Priss:96,priss:98,GalnPez2014DiscoveringNS,val:05,sporleder:02,petersen:04} and unstructured text corpora~\cite{hearst1992hyponyms,le2019inferring,cimiano_learning_2005,Elzinga:09,cigarran_twitter_fca_2016,Aznag2014LeveragingFC,wang_topic_feature_2021,wang_topic_hierarchy_2023,hirth_ordinal_2024,gutekunst2025conceptual}.
}

\begin{columns}[T,onlytextwidth]
\begin{column}{0.48\textwidth}
\textbf{Manual Annotation}
\begin{itemize}
    \item[+]<2-> Trust-worthy expert knowledge
    \item[-]<3-> Snapshot of evolving domain knowledge
    \item[-]<4-> costly, time-consuming
\end{itemize}
\end{column}

\begin{column}{0.48\textwidth}
\textbf{Data-Driven}
\begin{itemize}
    \item[-]<2-> Quality depends on data
    \item[+]<3-> Extends coverage of domains
    \item[+]<4-> Less costly, less time-consuming
\end{itemize}
\end{column}
\end{columns}

\vspace{0.5em}
\uncover<5->{
\begin{alertblock}{Problem}
    \begin{itemize}
        \item Does the data-driven hierarchy respect the expert knowledge? 
        \item How can we systematically compare the two hierarchies?
    \end{itemize}
\end{alertblock}
}

    \begin{block}{Task 2}

\end{frame}
